```python
def list_expenses(args):

    if args.from_date:

    if args.to_date:

        filtered = [e for e in filtered if datetime.date.fromisoformat(e["date"]) <= to_date]

    if not filtered:
        print("No expenses match the criteria.")
        return 0

    # Sort by date descending
    filtered.sort(key=lambda x: x["date"], reverse=True)
    for e in filtered:
        print(f"{e['date']} | {e['category']:10} | ${e['amount']:6.2f} | {e['description']}")
    return 0
```

```python
def summary(args):
    expenses = load_expenses()
    if not expenses:
        print("No expenses recorded.")
        return 0

    by = args.by or "category"
    groups = {}
    for e in expenses:
        key = e[by]
        groups[key] = groups.get(key, {"count": 0, "total": 0})
        groups[key]["count"] += 1
        groups[key]["total"] += e["amount"]

    print(f"\nSummary by {by.title()}:")
    print("-" * 50)
    for k, v in sorted(groups.items(), key=lambda x: x[1]["total"], reverse=True):
        print(f"{k:12} | {v['count']:2} entries | ${v['total']:6.2f}")
    return 0
```
