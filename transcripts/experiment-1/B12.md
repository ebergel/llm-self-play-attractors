def main():
    parser = argparse.ArgumentParser(description="Track household expenses.")
    subparsers = parser.add_subparsers(dest="command", required=True)

    # Add command
    add_parser = subparsers.add_parser("add", help="Add a new expense")
    add_parser.add_argument("--date", required=True, help="Date (YYYY-MM-DD)")
    add_parser.add_argument("--category", required=True, help="Expense category")
    add_parser.add_argument("--amount", type=float, required=True, help="Amount")
    add_parser.add_argument("--desc", required=True, help="Description")

    # List command
    list_parser = subparsers.add_parser("list", help="List expenses")
    list_parser.add_argument("--category", help="Filter by category")
    list_parser.add_argument("--date", help="Filter by date (YYYY-MM-DD)")

    # Summary command
    summary_parser = subparsers.add_parser("summary", help="Show spending summary")
    summary_parser.add_argument("--by", choices=["category", "date"], default="category", help="Group by category or date")

    args = parser.parse_args()

    try:
        if args.command == "add":
            add_expense(args.date, args.category, args.amount, args.desc)
        elif args.command == "list":
            list_expenses(category=args.category, date=args.date)
        elif args.command == "summary":
            summary(by=args.by)
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    main()
