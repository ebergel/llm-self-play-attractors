def validate_date(s):
    try:
        datetime.strptime(s, "%Y-%m-%d")
        return True
    except ValueError:
        raise argparse.ArgumentTypeError("Date must be YYYY-MM-DD")

def validate_amount(x):
    try:
        amt = float(x)
        if amt <= 0:
            raise ValueError
        return amt
    except ValueError:
        raise argparse.ArgumentTypeError("Amount must be positive number")

def load_data():
    with DATA_PATH.open() as f:
        return json.load(f)

def save_data(data):
    with DATA_PATH.open("w") as f:
        json.dump(data, f, indent=2)

def add(args):
    data = load_data()
    new_id = max([e["id"] for e in data], default=0) + 1
    expense = {
        "id": new_id,
        "amount": validate_amount(args.amount),
        "category": args.category,
        "date": validate_date(args.date),
        "description": args.description or ""
    }
    data.append(expense)
