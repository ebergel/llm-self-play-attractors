    data = ensure_data_file()
    totals = {}
    for e in data:
        cat = e["category"]
        typ = e["type"]
        totals.setdefault(cat, {"expense": 0, "income": 0})
        totals[cat][typ] += e["amount"]
    
    if args.format == "json":
        print(json.dumps(totals, indent=2))
    else:
        print("\nSummary by Category:")
        for cat in sorted(totals.keys()):
            exp = totals[cat]["expense"]
            inc = totals[cat]["income"]
            print(f"  {cat}: Expense ${exp:.2f}, Income ${inc:.2f}")
