def main():
    parser = argparse.ArgumentParser(description="Track household expenses.")
    subparsers = parser.add_subparsers(dest="command", required=True)

    # Add
    add = subparsers.add_parser("add", help="Add an expense")
    add.add_argument("--date", required=True, help="YYYY-MM-DD")
    add.add_argument("--category", required=True, help="Expense category")
    add.add_argument("--amount", type=float, required=True, help="Amount")
    add.add_argument("--desc", required=True, help="Description")

    # List
    list_parser = subparsers.add_parser("list", help="List expenses")
    list_parser.add_argument("--date")
    list_parser.add_argument("--category")

    # Total
    total_parser = subparsers.add_parser("total", help="Show total spend")
    total_parser.add_argument("--category")

    # Export
    subparsers.add_parser("export", help="Export to CSV")

    args = parser.parse_args()

    try:
        expenses = load_expenses()
        if args.command == "add":
            add_cmd(expenses, args)
            save_expenses(expenses)
        elif args.command == "list":
            list_cmd(expenses, args)
        elif args.command == "total":
            total_cmd(expenses, args)
        elif args.command == "export":
            export_cmd(expenses)
    except Exception as e:
        print(f"Error: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()

**storage.py** (draft):

import json
import os
from pathlib import Path
from typing import List, Dict, Any

EXPENSES_FILE = "expenses.json"

def load_expenses() -> List[Dict[str, Any]]:
    """Load expenses from JSON file. Returns empty list if file doesn't exist."""
    if not Path(EXPENSES_FILE).is_file():
        return []
    try:
        with open(EXPENSES_FILE, "r") as f:
            data = json.load(f)
        # Validate schema
        for entry in data:
            assert "date" in entry and isinstance(entry["date"], str)
            assert "category" in entry and isinstance(entry["category"], str)
            assert "amount" in entry and isinstance(entry["amount"], (int, float))
            assert entry["amount"] > 0
            assert "description" in entry and isinstance(entry["description"], str)
        return data
    except (json.JSONDecodeError, AssertionError, KeyError) as e:
        raise ValueError(f"Invalid expense data in {EXPENSES_FILE}: {e}")

def save_expenses(expenses: List[Dict[str, Any]]) -> None:
    """Save expenses to JSON file."""
    try:
        with open(EXPENSES_FILE, "w") as f:
            json.dump(expenses, f, indent=2)
    except Exception as e:
        raise IOError(f"Failed to save expenses: {e}")
```

**Notes**:  
- Uses `pathlib.Path` for robust file checks.  
- Validates schema on load: required fields, type, positive amount.  
- Raises clear errors for malformed data.  
- `json.dump` with `indent=2` for readability.

from typing import List, Dict, Any
from utils import error, format_csv

def add_cmd(expenses: List[Dict[str, Any]], args) -> None:
    """Add a new expense."""
    expense = {
        "date": args.date,
        "category": args.category,
        "amount": args.amount,
        "description": args.desc
    }
    expenses.append(expense)

def list_cmd(expenses: List[Dict[str, Any]], args) -> None:
    """List expenses filtered by date or category."""
    filtered = expenses
    if args.date:
        filtered = [e for e in filtered if e["date"] == args.date]
    if args.category:
        filtered = [e for e in filtered if e["category"] == args.category]
    # Sort by date
    sorted_expenses = sorted(filtered, key=lambda x: x["date"])
    for e in sorted_expenses:
        print(f"{e['date']} | {e['category']} | {e['amount']} | {e['description']}")

def total_cmd(expenses: List[Dict[str, Any]], args) -> None:
    """Show total spend, optionally by category."""
    total = 0.0
    if args.category:
        total = sum(e["amount"] for e in expenses if e["category"] == args.category)
    else:
        total = sum(e["amount"] for e in expenses)
    print(f"Total: {total:.2f}")

def export_cmd(expenses: List[Dict[str, Any]]) -> None:
    """Export to CSV."""
    print("date,category,amount,description")
    for e in expenses:
        print(f"{e['date']},{e['category']},{e['amount']},{e['description']}")

import sys
from datetime import date
from typing import List, Dict

def parse_date(s: str) -> date:
    """Parse YYYY-MM-DD string into date object."""
    try:
        return date.fromisoformat(s)
    except ValueError:
        error(f"Invalid date format. Use YYYY-MM-DD.")
        sys.exit(1)

def format_csv(expenses: List[Dict[str, Any]]) -> str:
    """Format expenses as CSV string."""
    lines = ["date,category,amount,description"]
    for e in expenses:
        lines.append(f"{e['date']},{e['category']},{e['amount']},{e['description']}")
    return "\n".join(lines)

def error(msg: str) -> None:
    """Print error to stderr."""
    print(f"Error: {msg}", file=sys.stderr)
```

**Notes**:  
- `parse_date` validates ISO format and exits on error.  
- `format_csv` returns a string for reuse (e.g., in `export_cmd` or tests).  
- `error` prints to stderr, consistent with CLI best practices.
